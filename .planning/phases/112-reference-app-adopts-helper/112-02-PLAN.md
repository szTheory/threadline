  Repo,
  [
    audit_context: audit_context,
    action: :post_created_via_api,
    transaction_meta: audit_transaction_meta(opts)
  ],
  fn ->
    case Repo.insert(Post.changeset(%Post{}, attrs)) do

    end
  end
