```elixir
# Bulk seed via Repo.insert_all/3 — much faster than 5_001+ individual seed_change!/1 calls.
# Pattern source: RESEARCH §P-10 lines 1004-1033.
defp bulk_seed_changes!(n, opts) when n > 0 and is_list(opts) do
  alias Threadline.Capture.{AuditChange, AuditTransaction}

  table = Keyword.fetch!(opts, :table)

  txn =
    @repo.insert!(
      AuditTransaction.changeset(%AuditTransaction{}, %{
        txid: :rand.uniform(1_000_000_000),
        occurred_at: DateTime.utc_now()
      })
    )

  now = DateTime.utc_now() |> DateTime.truncate(:microsecond)

  changes =
    for i <- 1..n do
      %{
        id: Ecto.UUID.generate(),
        transaction_id: txn.id,
        table_schema: "public",
        table_name: table,
        table_pk: %{"id" => "#{i}"},
        op: "insert",
        data_after: %{"i" => i},
        captured_at: now
      }
    end

  @repo.insert_all(AuditChange, changes)
end
```
